FROM python:3.11-slim

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=1

WORKDIR /app

# Install requirements (check build logs to ensure this runs!)
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy the source
COPY . .

# Run the bot
CMD ["python", "-m", "questbot.bot"]
