FROM python:3.11-slim

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=1

WORKDIR /app

# 1. install requirements first (cached layer)
COPY requirements.txt ./
RUN pip install --no-cache-dir -r requirements.txt

# 2. copy source (adjust paths if your bot is in a folder)
COPY . .

# 3. run directly (no entrypoint script needed)
CMD ["python", "-m", "questbot.bot"]
